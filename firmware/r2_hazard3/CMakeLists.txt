cmake_minimum_required(VERSION 3.13)

option(R2_HAZARD3_EMULATE "Arm harness smoke build; timings are not Hazard3" OFF)
set(PICO_BOARD pimoroni_pico_plus2_rp2350 CACHE STRING "")
if(R2_HAZARD3_EMULATE)
    set(PICO_PLATFORM rp2350-arm-s CACHE STRING "")
else()
    set(PICO_PLATFORM rp2350-riscv CACHE STRING "")
endif()
include(../pico_sdk_import.cmake)
project(r2_hazard3 C CXX ASM)
pico_sdk_init()

# Compile the exact C3 mapper twice; only the candidate gets the new helpers.
add_library(r2_map_base OBJECT ../src/dvbs2_8psk.c)
add_library(r2_map_zip OBJECT ../src/dvbs2_8psk.c)
target_compile_definitions(r2_map_zip PRIVATE IQ_R2_HAZARD3=1)
foreach(fn arena_words init frame flush codes_begin codes_step codes_frame codes_flush)
    target_compile_definitions(r2_map_base PRIVATE dvbs2_8psk_${fn}=r2_base_${fn})
endforeach()
add_executable(r2map main.c $<TARGET_OBJECTS:r2_map_base> $<TARGET_OBJECTS:r2_map_zip>)
foreach(target r2_map_base r2_map_zip r2map)
    target_include_directories(${target} PRIVATE ../src)
    target_compile_definitions(${target} PRIVATE IQ_ON_DEVICE=1 PICO_STACK_SIZE=0x1000)
    target_compile_options(${target} PRIVATE -O3 -Wall -Wextra -Werror)
    if(R2_HAZARD3_EMULATE)
        target_compile_definitions(${target} PRIVATE IQ_R2_HAZARD3_EMULATE=1)
    endif()
endforeach()
target_link_libraries(r2map pico_stdlib hardware_clocks)
pico_enable_stdio_usb(r2map 1)
pico_enable_stdio_uart(r2map 0)
pico_add_extra_outputs(r2map)
